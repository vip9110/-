cmake_minimum_required(VERSION 3.20)
project(LiliDesktopPet VERSION 3.0.0 LANGUAGES CXX RC)

option(LILI_BUILD_TESTS "Build portable unit tests" ON)

if(WIN32)
  add_executable(LiliDesktopPet WIN32
    src/main.cpp src/app.cpp src/pet_window.cpp src/panel.cpp src/system.cpp src/sprites.cpp src/settings.cpp src/pet.rc)
  target_include_directories(LiliDesktopPet PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src" "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_features(LiliDesktopPet PRIVATE cxx_std_17)
  target_compile_definitions(LiliDesktopPet PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(LiliDesktopPet PRIVATE gdiplus ole32 oleaut32 shell32 user32 gdi32 uuid advapi32)
  set_target_properties(LiliDesktopPet PROPERTIES OUTPUT_NAME "LiliDesktopPet")
  if(MINGW)
    target_compile_options(LiliDesktopPet PRIVATE -Wall -Wextra)
    target_link_options(LiliDesktopPet PRIVATE -municode -static -static-libgcc -static-libstdc++)
  elseif(MSVC)
    target_compile_options(LiliDesktopPet PRIVATE /utf-8 /W4)
    set_property(TARGET LiliDesktopPet PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
endif()

if(LILI_BUILD_TESTS AND NOT CMAKE_CROSSCOMPILING)
  enable_testing()
  foreach(name logic animation motion_flow blit)
    add_executable(test_${name} tests/test_${name}.cpp)
    target_compile_features(test_${name} PRIVATE cxx_std_17)
    if(MSVC)
      target_compile_options(test_${name} PRIVATE /utf-8)
    endif()
    add_test(NAME ${name} COMMAND test_${name} "${CMAKE_CURRENT_SOURCE_DIR}/assets/motion-vectors.bin")
  endforeach()
endif()
